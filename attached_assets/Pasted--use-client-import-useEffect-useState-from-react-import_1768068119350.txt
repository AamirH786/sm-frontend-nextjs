"use client";

import { useEffect, useState } from "react";
import { useForm, Controller } from "react-hook-form";
import { Edit, Trash2, Plus } from "lucide-react";
import Swal from "sweetalert2";

import permissionService from "@/services/user-management/permissionService";
import moduleService from "@/services/user-management/moduleService";
import actionService from "@/services/user-management/actionService";

import { useToast } from "@/context/ToastContext";
import { Card, CardContent } from "@/components/ui/Card";
import Button from "@/components/ui/Button";
import Modal from "@/components/ui/Modal";
import Select from "@/components/ui/Select";
import AdvancedDataTable from "@/components/ui/AdvancedDataTable";
import Input from "@/components/ui/Input";
import { formatDateTime } from "@/lib/utils";
import { showApiError } from "@/lib/showError";
import Tooltip from "@/components/ui/Tooltip";

import HasPermission from "@/components/permissions/HasPermission";
import usePermission from "@/hooks/usePermission";
import { useAuth } from "@/hooks/useAuth";
import { useRouter } from "next/navigation";

export default function PermissionsPage() {
  const router = useRouter();
  const { user, isLoading } = useAuth();
  const { can } = usePermission();
  const { showToast } = useToast();

  const [modules, setModules] = useState<any[]>([]);
  const [actions, setActions] = useState<any[]>([]);

  const [showFilters, setShowFilters] = useState(false);
  const [refreshKey, setRefreshKey] = useState(0);
  const [isModalOpen, setIsModalOpen] = useState(false);
  const [editing, setEditing] = useState<any>(null);

  const [filters, setFilters] = useState({
    module_id: "",
    action_id: "",
    status: "",
    search: "",
  });

  const reloadTable = () => setRefreshKey((k) => k + 1);

  /* -----------------------------------------------------
      PAGE VIEW PERMISSION
  ----------------------------------------------------- */
  useEffect(() => {
    if (isLoading) return;
    if (!can("um_permissions", "view")) {
      router.replace("/403");
      return;
    }
  }, [isLoading, can]);

  /* LOAD DROP-DOWNS */
  useEffect(() => {
    loadModules();
    loadActions();
  }, []);

  const loadModules = async () => {
    const res: any = await moduleService.getModuleList();
    setModules(res.data);
  };

  const loadActions = async () => {
    const res: any = await actionService.dropdown();
    setActions(res.data);
  };

  useEffect(() => {
    reloadTable();
  }, [filters.status, filters.module_id, filters.action_id]);

  /* ---------------- FORM ---------------- */
  const {
    register,
    handleSubmit,
    reset,
    watch,
    setValue,
    control,
    formState: { errors },
  } = useForm();

  const handleOpenCreate = () => {
    setEditing(null);
    reset({
      module_id: "",
      action_id: "",
      description: "",
      is_active: 1,
    });
    setIsModalOpen(true);
  };

  const handleOpenEdit = (row: any) => {
    setEditing(row);
    reset({
      module_id: row.module_id,
      action_id: row.action_id,
      description: row.description,
      is_active: row.is_active,
    });
    setIsModalOpen(true);
  };

  /* ---------------- DELETE ---------------- */
  const handleDelete = async (id: number) => {
    const confirm = await Swal.fire({
      title: "Delete Permission?",
      text: "You cannot undo this action.",
      icon: "warning",
      showCancelButton: true,
      confirmButtonColor: "#b91c1c",
      cancelButtonColor: "#6b7280",
      confirmButtonText: "Delete",
    });

    if (!confirm.isConfirmed) return;

    try {
      await permissionService.delete(id);
      showToast("Permission deleted", "success");
      reloadTable();
    } catch (error: any) {
      showToast(error.message, "error");
    }
  };

  /* ---------------- SUBMIT ---------------- */
  const onSubmit = async (data: any) => {
    try {
      if (editing) {
        await permissionService.update(editing.id, data);
        showToast("Permission updated successfully", "success");
      } else {
        await permissionService.create(data);
        showToast("Permission created successfully", "success");
      }

      setIsModalOpen(false);
      reset();
      reloadTable();
    } catch (error: any) {
      showApiError(error.message, showToast);
    }
  };

  /* ---------------- TABLE ---------------- */
  const fetchTable = async ({ page, limit, search, sortBy, sortOrder }: any) => {
    const params: any = { page, limit };

    if (search) params.search = search;
    if (filters.module_id) params.module_id = filters.module_id;
    if (filters.action_id) params.action_id = filters.action_id;
    if (filters.status !== "") params.is_active = Number(filters.status);

    if (sortBy) params.sortBy = sortBy;
    if (sortOrder) params.sortOrder = sortOrder;

    const res = await permissionService.list(params);

    return {
      data: res.data,
      pagination: res.pagination,
      analytics: {
        totalActive: res.data.filter((x: any) => x.is_active === 1).length,
        totalInactive: res.data.filter((x: any) => x.is_active === 0).length,
      },
    };
  };

  const isFilterApplied =
    filters.module_id !== "" ||
    filters.action_id !== "" ||
    filters.status !== "" ||
    filters.search.trim() !== "";

  /* ---------------- UI ---------------- */
  return (
    <div>
      {/* HEADER */}
      <div className="flex justify-between items-center mb-5">
        <h1 className="text-2xl font-bold">Permissions</h1>

        <div className="flex gap-3">

          {/* ⭐ RESET FILTERS */}
          {isFilterApplied && (
            <Button
              variant="outline"
              onClick={() =>
                setFilters({
                  module_id: "",
                  action_id: "",
                  status: "",
                  search: "",
                })
              }
              className="flex items-center gap-2 border border-red-300/60 text-red-600 hover:bg-red-50 rounded-lg px-3 py-2"
            >
              <svg xmlns="http://www.w3.org/2000/svg" width="16" height="16" fill="none" viewBox="0 0 24 24" stroke="currentColor" strokeWidth="2">
                <line x1="18" y1="6" x2="6" y2="18" />
                <line x1="6" y1="6" x2="18" y2="18" />
              </svg>
              Reset Filters
            </Button>
          )}

          {/* FILTERS */}
          <Button
            variant="outline"
            onClick={() => setShowFilters((p) => !p)}
            className="flex items-center gap-2 border border-gray-300 hover:bg-gray-100 text-gray-700 rounded-lg px-4 py-2 shadow-sm"
          >
            <svg
              xmlns="http://www.w3.org/2000/svg"
              width="18"
              height="18"
              fill="none"
              viewBox="0 0 24 24"
              stroke="currentColor"
              strokeWidth="2"
              className="shrink-0"
            >
              <polygon points="22 3 2 3 10 12 10 19 14 21 14 12 22 3" />
            </svg>
            Filters
          </Button>

          {/* ADD PERMISSION */}
          <HasPermission module="um_permissions" action="add">
            <Button onClick={handleOpenCreate}>
              <Plus size={18} className="mr-2" /> Add Permission
            </Button>
          </HasPermission>

        </div>
      </div>

      {/* FILTERS */}
      {showFilters && (
        <div className="grid grid-cols-1 sm:grid-cols-3 gap-4 mb-6">

          <Select
            label="Filter by Module"
            value={filters.module_id}
            options={[
              { value: "", label: "All" },
              ...modules.map((m: any) => ({
                value: m.id,
                label: `${m.module_group ? m.module_group + " → " : ""}${
                  m.module_name
                }`,
              })),
            ]}
            onChange={(val) =>
              setFilters((prev) => ({ ...prev, module_id: String(val) }))
            }
          />

          <Select
            label="Filter by Action"
            value={filters.action_id}
            options={[
              { value: "", label: "All" },
              ...actions.map((a: any) => ({
                value: a.id,
                label: a.action_name,
              })),
            ]}
            onChange={(val) =>
              setFilters((prev) => ({ ...prev, action_id: String(val) }))
            }
          />

          <Select
            label="Filter by Status"
            value={filters.status}
            options={[
              { value: "", label: "All" },
              { value: "1", label: "Active" },
              { value: "0", label: "Inactive" },
            ]}
            onChange={(val) =>
              setFilters((prev) => ({ ...prev, status: String(val) }))
            }
          />
        </div>
      )}

      {/* TABLE */}
      <Card>
        <CardContent>
          <AdvancedDataTable
            refreshTrigger={refreshKey}
            externalSearch={filters.search}         // ⭐ ADDED
            onSearchChange={(v) =>
              setFilters((prev) => ({ ...prev, search: v }))
            }                                        // ⭐ ADDED
            columns={[
              { key: "permission_name", label: "Permission Name" },
              { key: "permission_slug", label: "Slug" },

              {
                key: "module",
                label: "Module",
                render: (r: any) => r.module?.module_name || "-",
              },

              {
                key: "action",
                label: "Action",
                render: (r: any) => r.action?.action_name || "-",
              },

              {
                key: "is_active",
                label: "Status",
                render: (r: any) =>
                  r.is_active ? (
                    <span className="text-green-600 font-medium">Active</span>
                  ) : (
                    <span className="text-red-600 font-medium">Inactive</span>
                  ),
              },

              {
                key: "created_at",
                label: "Created",
                render: (r: any) =>
                  r.created_at ? formatDateTime(r.created_at) : "-",
              },
            ]}
            fetchData={fetchTable}
            showAnalytics
            actions={(row: any) => (
              <div className="flex justify-end gap-2">

                <HasPermission module="um_permissions" action="update">
                  <Tooltip text="Edit">
                    <button
                      onClick={() => handleOpenEdit(row)}
                      className="p-1.5 hover:bg-gray-100 rounded"
                    >
                      <Edit size={16} />
                    </button>
                  </Tooltip>
                </HasPermission>

                <HasPermission module="um_permissions" action="delete">
                  <Tooltip text="Delete">
                    <button
                      onClick={() => handleDelete(row.id)}
                      className="p-1.5 text-red-600 hover:bg-red-50 rounded"
                    >
                      <Trash2 size={16} />
                    </button>
                  </Tooltip>
                </HasPermission>

              </div>
            )}
          />
        </CardContent>
      </Card>

      {/* MODAL */}
      <Modal
        isOpen={isModalOpen}
        onClose={() => {
          setIsModalOpen(false);
          setEditing(null);
          reset();
        }}
        title={editing ? "Edit Permission" : "Add Permission"}
      >
        <form onSubmit={handleSubmit(onSubmit)} className="space-y-4" noValidate>

          <Controller
            name="module_id"
            control={control}
            rules={{ required: "Module is required" }}
            render={({ field, fieldState }) => (
              <Select
                label="Module"
                value={field.value ?? ""}
                options={[
                  // { value: "", label: "Select Module" },
                  ...modules.map((m: any) => ({
                    value: m.id,
                    label: `${m.module_group ? m.module_group + " → " : ""}${
                      m.module_name
                    }`,
                  })),
                ]}
                onChange={field.onChange}
                required
                error={fieldState.error?.message}
              />
            )}
          />

          <Controller
            name="action_id"
            control={control}
            rules={{ required: "Action is required" }}
            render={({ field, fieldState }) => (
              <Select
                label="Action"
                value={field.value ?? ""}
                onChange={field.onChange}
                options={actions.map((a: any) => ({
                  value: a.id,
                  label: a.action_name,
                }))}
                required
                error={fieldState.error?.message}
              />
            )}
          />

          <Input
            label="Description"
            {...register("description")}
            error={errors.description?.message}
          />

          <Select
            label="Status"
            value={watch("is_active")}
            options={[
              { value: 1, label: "Active" },
              { value: 0, label: "Inactive" },
            ]}
            onChange={(val) => setValue("is_active", val)}
          />

          <div className="flex justify-end gap-2 pt-2">
            <Button
              type="button"
              variant="secondary"
              onClick={() => {
                setIsModalOpen(false);
                setEditing(null);
                reset();
              }}
            >
              Cancel
            </Button>

            <Button type="submit">
              {editing ? "Update" : "Create"}
            </Button>
          </div>
        </form>
      </Modal>
    </div>
  );
}
