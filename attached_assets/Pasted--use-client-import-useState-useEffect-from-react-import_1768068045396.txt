"use client";

import { useState, useEffect } from "react";
import boppService from "@/services/masters/boppService";
import { Card, CardContent } from "@/components/ui/Card";
import Button from "@/components/ui/Button";
import Input from "@/components/ui/Input";
import Modal from "@/components/ui/Modal";
import Swal from "sweetalert2";
import { Edit, Trash2, Plus } from "lucide-react";
import { useToast } from "@/context/ToastContext";
import AdvancedDataTable from "@/components/ui/AdvancedDataTable";
import Select from "@/components/ui/Select";
import { showApiError } from "@/lib/showError";
import { Controller, useForm } from "react-hook-form";
import { formatDateTime } from "@/lib/utils";
import Tooltip from "../ui/Tooltip";

import usePermission from "@/hooks/usePermission";
import HasPermission from "@/components/permissions/HasPermission";
import { useRouter } from "next/navigation";

export default function BoppItemList() {
  const { showToast } = useToast();
  const { can, isLoading } = usePermission();
  const router = useRouter();

  const [isModalOpen, setIsModalOpen] = useState(false);
  const [editing, setEditing] = useState<any>(null);
  const [categories, setCategories] = useState<any[]>([]);
  const [showFilters, setShowFilters] = useState(false);
  const [refreshKey, setRefreshKey] = useState(0);
  const reloadTable = () => setRefreshKey((k) => k + 1);


  const [filters, setFilters] = useState({
    category: "", 
    status: "",
    search: "",
  });

  /* --------------------------------------------------------
      ⭐ VIEW PERMISSION GUARD
  -------------------------------------------------------- */
  useEffect(() => {
    if (isLoading) return;     // wait for auth to load
    const allowed = can("bopp_items", "view");
    if (allowed === false) router.replace("/403");
  }, [isLoading, can]);

  /* ------------------------------ React Hook Form ------------------------------ */
  const { register, handleSubmit, reset, setValue, watch, control, formState: { errors } } = useForm();

  /* ------------------------------ LOAD CATEGORIES ------------------------------ */
  useEffect(() => {
    loadCategories();
  }, []);



  const loadCategories = async () => {
    try {
      const res = await boppService.listCategories({ page: 1, limit: 200 });
      setCategories(res.data);
    } catch { }
  };

  /* ------------------------------ FILTER CHANGE → Reload Table ------------------------------ */
  useEffect(() => {
    reloadTable();
  }, [filters.category, filters.status]);

  /* ------------------------------ OPEN CREATE ------------------------------ */
  const handleOpenCreate = () => {
    setEditing(null);
    reset({
      item_code: "",
      bopp_size: "",
      // ❌ Removed bopp_category from reset
      bopp_category_id: "", // New ID field
      bopp_micron: "",
      status: 1,
    });
    setIsModalOpen(true);
  };

  /* ------------------------------ OPEN EDIT ------------------------------ */
  const handleOpenEdit = (row: any) => {
    setEditing(row);

    reset({
      item_code: row.item_code,
      bopp_size: row.bopp_size,
      // ❌ Removed bopp_category from reset
      bopp_category_id: row.bopp_category_id || "", // Set new category ID
      bopp_micron: row.bopp_micron,
      status: row.status,
    });

    setIsModalOpen(true);
  };

  /* ------------------------------ DELETE ------------------------------ */
  const handleDelete = async (id: number) => {
    const confirm = await Swal.fire({
      title: "Delete Item?",
      text: "You cannot undo this action.",
      icon: "warning",
      showCancelButton: true,
      confirmButtonColor: "#b91c1c",
      cancelButtonColor: "#6b7280",
      confirmButtonText: "Delete",
    });

    if (!confirm.isConfirmed) return;

    try {
      await boppService.deleteItem(id);
      showToast("Item deleted", "success");
      reloadTable();
    } catch (err: any) {
      showToast(err.message, "error");
    }
  };

  /* ------------------------------ SUBMIT ------------------------------ */
  const onSubmit = async (data: any) => {
    // ❌ Removed complex logic, now only send ID
    const payload = {
      ...data,
      bopp_category_id: data.bopp_category_id ? Number(data.bopp_category_id) : null,
    };

    try {
      if (editing) {
        await boppService.updateItem(editing.id, payload);
        showToast("Item updated successfully", "success");
      } else {
        await boppService.createItem(payload);
        showToast("Item created successfully", "success");
      }

      setIsModalOpen(false);
      reset();
      reloadTable();
    } catch (err: any) {
      showApiError(err.message, showToast);
    }
  };

  /* ------------------------------ TABLE FETCH ------------------------------ */
  const fetchTable = async ({ page, limit, search, sortBy, sortOrder }: any) => {
    const params: any = { page, limit };

    if (search) params.search = search;
    if (filters.category) params.category = filters.category; 
    if (filters.status !== "") params.status = filters.status;
    if (sortBy) params.sortBy = sortBy;
    if (sortOrder) params.sortOrder = sortOrder;

    const res = await boppService.listItems(params);

    return {
      data: res.data,
      pagination: res.pagination,
      analytics: {
        totalActive: res.data.filter((x: any) => x.status === 1).length,
        totalInactive: res.data.filter((x: any) => x.status === 0).length,
      },
    };
  };

  const isFilterApplied =
    filters.category !== "" ||
    filters.status !== "" ||
    (filters.search && filters.search.trim() !== "");

  /* ------------------------------ UI ------------------------------ */
  return (
    <div>
      {/* PAGE HEADER */}
      <div className="flex items-center justify-between mb-5">
        <h1 className="text-2xl font-bold">BOPP Items</h1>

        <div className="flex items-center gap-3">
          {/* FILTER BUTTON */}
          {isFilterApplied && (

            <Button
              variant="outline"
              onClick={() => {
                setFilters({ category: "", status: "", search: "" });
              }}
              className="flex items-center gap-2 border border-red-300/60 text-red-600 hover:bg-red-50 
                        rounded-lg px-3 py-2 transition-all"
            >
              <svg
                xmlns="http://www.w3.org/2000/svg"
                width="16"
                height="16"
                viewBox="0 0 24 24"
                fill="none"
                stroke="currentColor"
                strokeWidth="2"
                strokeLinecap="round"
                strokeLinejoin="round"
                className="text-red-600"
              >
                <line x1="18" y1="6" x2="6" y2="18" />
                <line x1="6" y1="6" x2="18" y2="18" />
              </svg>

              <span className="text-sm font-medium">Reset Filters</span>
            </Button>


          )}
          <Button
            variant="outline"
            onClick={() => setShowFilters((p) => !p)}
            className="flex items-center gap-2 border border-gray-300 hover:bg-gray-100 text-gray-700 rounded-lg px-4 py-2 shadow-sm"
          >
            <svg
              xmlns="http://www.w3.org/2000/svg"
              width="18"
              height="18"
              viewBox="0 0 24 24"
              fill="none"
              stroke="currentColor"
              strokeWidth="2"
              strokeLinecap="round"
              strokeLinejoin="round"
            >
              <polygon points="22 3 2 3 10 12 10 19 14 21 14 12 22 3" />
            </svg>
            Filters
          </Button>

          {/* ⭐ ADD ITEM BUTTON with permission */}
          <HasPermission module="bopp_items" action="add">
            <Button onClick={handleOpenCreate}>
              <Plus size={18} className="mr-2" />
              Add Item
            </Button>
          </HasPermission>
        </div>
      </div>

      {/* FILTERS */}
      {showFilters && (
        <div className="grid grid-cols-1 sm:grid-cols-3 gap-4 mb-6 animate-fadeIn">
          <Select
            label="Filter by Category"
            value={filters.category}
            options={[
              { value: "", label: "All" },
              ...categories.map((c: any) => ({
                value: c.id, 
                label: c.category_name,
              })),
            ]}
            onChange={(val) =>
              setFilters((prev) => ({ ...prev, category: String(val) }))
            }
          />

          <Select
            label="Filter by Status"
            value={filters.status}
            options={[
              { value: "", label: "All" },
              { value: "1", label: "Active" },
              { value: "0", label: "Inactive" },
            ]}
            onChange={(val) =>
              setFilters((prev) => ({ ...prev, status: String(val) }))
            }
          />
        </div>
      )}

      {/* TABLE */}
      <Card>
        <CardContent>
          <AdvancedDataTable
            refreshTrigger={refreshKey}
            externalSearch={filters.search}
            onSearchChange={(val) =>
              setFilters((prev) => ({ ...prev, search: val }))
            }
            columns={[
              { key: "item_code", label: "Item Code" },
              { key: "bopp_size", label: "Size" },
              { key: "bopp_micron", label: "Micron" },
              { key: "bopp_category", label: "Category Code" }, // ✅ Using bopp_category field populated by JOIN
              {
                key: "status",
                label: "Status",
                render: (r: any) =>
                  r.status === 1 ? (
                    <span className="text-green-600 font-medium">Active</span>
                  ) : (
                    <span className="text-red-600 font-medium">Inactive</span>
                  ),
              },
              {
                key: "created_at",
                label: "Created",
                render: (r: any) =>
                  r.created_at ? formatDateTime(r.created_at) : "-",
              },
            ]}
            fetchData={fetchTable}
            showAnalytics
            actions={(row: any) => (
              <div className="flex justify-end gap-2">

                {/* ⭐ EDIT */}
                <HasPermission module="bopp_items" action="update">
                  <Tooltip text="Edit">
                    <button
                      onClick={() => handleOpenEdit(row)}
                      className="p-1.5 hover:bg-gray-100 rounded"
                    >
                      <Edit size={16} />
                    </button>
                  </Tooltip>
                </HasPermission>

                {/* ⭐ DELETE */}
                <HasPermission module="bopp_items" action="delete">
                  <Tooltip text="Delete">
                    <button
                      onClick={() => handleDelete(row.id)}
                      className="p-1.5 text-red-600 hover:bg-red-50 rounded"
                    >
                      <Trash2 size={16} />
                    </button>
                  </Tooltip>
                </HasPermission>

              </div>
            )}
          />
        </CardContent>
      </Card>

      {/* MODAL */}
      <Modal
        isOpen={isModalOpen}
        onClose={() => {
          setIsModalOpen(false);
          setEditing(null);
          reset();
        }}
        title={editing ? "Edit Item" : "Add Item"}
      >
        <form onSubmit={handleSubmit(onSubmit)} className="space-y-4" noValidate>
          <Input
            label="Item Code"
            {...register("item_code", { required: "Item code is required" })}
            error={errors.item_code?.message}
            required
          />

          <Input
            label="Size"
            type="text"
            inputMode="decimal"
            {...register("bopp_size", {
              required: "Size is required",
              pattern: {
                value: /^[0-9]*\.?[0-9]+$/,
                message: "Only numbers or decimal allowed",
              },
            })}
            error={errors.bopp_size?.message}
            required
          />

          <Input
            label="Micron"
            type="text"
            inputMode="decimal"
            {...register("bopp_micron", {
              required: "Micron is required",
              pattern: {
                value: /^[0-9]*\.?[0-9]+$/,
                message: "Only numbers or decimal allowed",
              },
            })}
            error={errors.bopp_micron?.message}
            required
          />

          <Controller
            name="bopp_category_id" 
            control={control}
            rules={{ required: "Category is required" }}
            render={({ field, fieldState }) => (
              <Select
                label="Category"
                value={field.value ?? ""}
                options={categories.map((c: any) => ({
                  value: c.id, 
                  label: c.category_name,
                }))}
                required
                onChange={(val) => field.onChange(val)}
                error={fieldState.error?.message}
              />
            )}
          />

          <Select
            label="Status"
            value={watch("status")}
            options={[
              { value: 1, label: "Active" },
              { value: 0, label: "Inactive" },
            ]}
            onChange={(val) => setValue("status", val)}
          />

          <div className="flex justify-end gap-2 pt-2">
            <Button
              type="button"
              variant="secondary"
              onClick={() => {
                setIsModalOpen(false);
                setEditing(null);
                reset();
              }}
            >
              Cancel
            </Button>

            <Button type="submit">
              {editing ? "Update" : "Create"}
            </Button>
          </div>
        </form>
      </Modal>
    </div>
  );
}
